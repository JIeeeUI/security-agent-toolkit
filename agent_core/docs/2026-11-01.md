        print(count)

counts = [0, 4, 2, 5]
selected = 0
total = 0
for count in counts:
    if count >=3:
        selected = selected + 1
        total = total + count
print(selected, total)

# 새로 알게 된 점
selected + 1 와 total + count 를 새롭게 배웠다
total 같은 경우에는 이해가 바로 됐다
하지만 selected 같은 경우에는 이해가 바로 안됨
AI한테 공부 중인 파이썬 대화에 물어본 결과
selected는 이름을 바꿀 수 있는 변수라고 해 줌
total도 같다고 볼 수 있다고 함
대입이냐고 물었을 때 아니라고 하면서 위와 같이 알려줌
